# This script calculates the vibrational information and total relaxed energy of different structures of CrV, TaV, TaW for a given configuration.

#(1) Initialization

units metal
boundary p p p
atom_style atomic
atom_modify map array

#(2) Import data

read_data configuration.dat

#(3) Definition of interatomic potential
pair_style eam/alloy
pair_coeff * * WTaCrV.eam.alloy.zbl Cr V

#(4) Functions

thermo 10
thermo_style custom step pe press pxx pyy pzz vol fmax fnorm

fix 1 all box/relax aniso 0.0 vmax 0.01
min_style cg
minimize 1.0e-10 1.0e-10 10000 100000
minimize 1.0e-10 1.0e-10 10000 100000
minimize 1.0e-10 1.0e-10 10000 100000
unfix 1
variable Relaxed_energy equal pe

dynamical_matrix all eskm 0.0001 file dynmat.dat

#(5) Simulation outputs
print "${Relaxed_energy}" file relaxed_energy.dat
